wget -O humandb/hg19_refGene.txt.gz \

# Download the gene locations
  http://hgdownload.cse.ucsc.edu/goldenPath/hg19/database/refGene.txt.gz
gunzip humandb/hg19_refGene.txt.gz

# Download the human genome fasta
wget -O humandb/hg19.fa.gz \
  http://hgdownload.cse.ucsc.edu/goldenPath/hg19/bigZips/hg19.fa.gz
gunzip humandb/hg19.fa.gz

# Generate hg19_refGeneMrna.fa (mRNA for each transcript for amino acid change annotation)
# This took a couple minutes
singularity exec --bind "$PWD:$PWD" annovar_2018Apr16.sif bash -lc '
cd "'"$PWD"'" &&
perl /home/TOOLS/tools/annovar/current/bin/retrieve_seq_from_fasta.pl \
  humandb/hg19_refGene.txt \
  -format refGene \
  -seqfile humandb/hg19.fa \
  -outfile humandb/hg19_refGeneMrna.fa
'

# Run annovar
singularity exec --bind "$PWD:$PWD" annovar_2018Apr16.sif bash -lc '
cd "'"$PWD"'" &&
perl /home/TOOLS/tools/annovar/current/bin/table_annovar.pl \
  ra_with_spikes.vcf \
  humandb/ \
  -buildver hg19 \
  -out example \
  -protocol refGene \
  -operation g \
  -nastring . \
  -vcfinput
'
