### "Exercise" 3 (curse of dimensionality)

In higher dimensions, points tend to become equidistant.

```{webr}
# 3a Distance concentration
# Draw 200 points from a normal distribution in d dimensions
# How close are the closest pair? How far are the furthest pair?
# Get the ratio of those distances (biggest to smallest)
set.seed(1)
distance_spread <- function(d, n=200) {
  X1 <- matrix(rnorm(n*d), nrow=n)
  D <- as.matrix(dist(X1))
  D <- D[upper.tri(D)]
  c(min=min(D), max=max(D), mean=mean(D))
}
# Do this in a bunch of dimensions from 2 to 10,000
dims <- c(2, 5, 10, 50, 100, 500, 1000, 10000)
results <- sapply(dims, distance_spread)
ratio <- results["min",] / results["max",]
plot(dims, ratio, type="b",
     xlab="Dimension",
     ylab="Min/max distance ratio",
     main="Distance concentration")
```

In higher dimensions, most of the volume is "near the boundary."

```{webr}
# 3b Volume explosion
# It's pretty obvious that measure increases exponentially with dimension
# But also see that most of the space is near the "boundary"
# (The unit ball is *inside* the unit cube in each space)
vol_ratio <- function(d) {
  (pi^(d/2) / gamma(d/2 + 1)) / (2^d)
}
dims <- 1:20
ratios <- sapply(dims, vol_ratio)
plot(dims, ratios, type="b",
     xlab="Dimension",
     ylab="Volume of unit ball / unit cube")
```

In higher dimensions, there is a lot of empty space.

```{webr}
# 3c Sparsity
# The average distance between uniform points gets bigger with dimension
dims <- 1:50
mean_dist <- sapply(dims, function(d) {
  X1 <- matrix(runif(1000*d), ncol=d)
  mean(dist(X1))
})
plot(dims, mean_dist, type="b",
     xlab="Dimension",
     ylab="Mean distance (uniform cube)")
```

In high-dimensional datasets, noise can look like structure.

```{webr}
# 3d "Hallucinating" structure
# Simulate random data
set.seed(1)
n <- 400
d <- 100
X1 <- matrix(rnorm(n*d), nrow=n)
# Apply fancy ML stuff!
# PCA for visualization
# K-means clustering in original high-D space
pca <- prcomp(X1, scale.=TRUE)
Z <- pca$x[,1:2]
# Plot the clusters
# Wow, we found patterns in the data! (But the data is random and meaningless)
km <- kmeans(X1, centers=4, nstart=50)
plot(Z, col=km$cluster, pch=16,
     xlab="PC1", ylab="PC2",
     main="Pure noise + k-means clustering")
# Draw convex hulls to exaggerate cluster appearance
for (k in 1:4) {
  pts <- Z[km$cluster == k, ]
  hull <- chull(pts)
  polygon(pts[hull, ], border=k, lwd=2)
}
```

Here's a function that will implement an interactive k-means algorithm to cluster the data. You don't need to change anything in this chunk. You'll run the `kmeans_click` function in the chunk after.

```{webr}
k <- 3
kmeans_click <- function(x, y, k = 3, iters = 20, seed = 1) {
  set.seed(seed)
  pts <- cbind(as.numeric(x), as.numeric(y))
  pts <- pts[complete.cases(pts), , drop = FALSE]
  n <- nrow(pts)
  
  cols <- c("red","blue","darkgreen","orange","purple","brown","cyan","magenta")[1:k]
  centers <- pts[sample.int(n, k), , drop = FALSE]
  
  # show initial random seeds
  plot(pts[,1], pts[,2], pch=16, col="grey70", xlab="X", ylab="Y",
       main="init seeds (click)")
  points(centers[,1], centers[,2], pch=4, cex=3, lwd=3, col=cols)
  locator(1)
  
  cl_old <- integer(n)
  
  for (t in 1:iters) {
    # assign
    d2 <- sapply(1:k, function(j)
      (pts[,1]-centers[j,1])^2 + (pts[,2]-centers[j,2])^2
    )
    cl <- max.col(-d2)
    
    # update (with empty-cluster reseed)
    for (j in 1:k) {
      idx <- which(cl == j)
      if (length(idx) > 0) centers[j,] <- colMeans(pts[idx, , drop=FALSE])
      else centers[j,] <- pts[sample.int(n, 1), ]
    }
    
    plot(pts[,1], pts[,2], pch=16, col=cols[cl], xlab="X", ylab="Y",
         main=paste("iter", t, "(click)"))
    points(centers[,1], centers[,2], pch=4, cex=3, lwd=3, col=cols)
    locator(1)
    
    if (identical(cl, cl_old)) break
    cl_old <- cl
  }
  
  invisible(list(cluster=cl, centers=centers))
}

```

### Exercise 10

Now try interactively running k-means on the principal components of ancestry contained in `pca$x`. This chunk just sets up the function you'll use - don't change it.

```{webr}
kmeans_pca_step_s3d <- function(pcx,
                                k = 3, iters = 20, seed = 1,
                                use_pcs = NULL,          # which PCs to use for distance; NULL = all
                                plot_pcs = 1:3,          # which 3 PCs to plot
                                pause = TRUE,
                                pch = 16, cex = 0.4,
                                center_pch = 4, center_cex = 2.2, center_lwd = 3,
                                theta = 40, phi = 20, truth=NULL) {

  if (!requireNamespace("scatterplot3d", quietly = TRUE)) {
    stop("Install scatterplot3d: install.packages('scatterplot3d')")
  }

  pcx <- as.matrix(pcx)
  keep <- complete.cases(pcx)
  pcx <- pcx[keep, , drop = FALSE]
  if (!is.null(truth)) truth <- truth[keep]
  n <- nrow(pcx)
  
  if (is.null(truth)) {
  pch_vec <- pch
} else {
  # map each truth group to a plotting character
  tt <- as.factor(truth)
  pchs <- c(1:25)                    # available base pch values
  if (nlevels(tt) > length(pchs)) stop("Too many truth groups for available pch values.")
  pch_vec <- pchs[as.integer(tt)]
}
  
  p <- ncol(pcx)
  if (n < k) stop("Need at least k complete rows in pcx.")
  if (p < 3) stop("Need at least 3 PCs to plot.")

  if (is.null(use_pcs)) use_pcs <- seq_len(p)
  if (length(plot_pcs) != 3) stop("plot_pcs must be length 3 (e.g., 1:3).")

  use_pcs <- as.integer(use_pcs)
  plot_pcs <- as.integer(plot_pcs)

  # data for distance + for plotting
  pts_use  <- pcx[, use_pcs, drop = FALSE]
  pts_plot <- pcx[, plot_pcs, drop = FALSE]

  set.seed(seed)

  cols_all <- c("red","blue","darkgreen","orange","purple","brown","cyan","magenta")
  if (k > length(cols_all)) stop("k too large for built-in color list.")
  cols <- cols_all[1:k]

  # init centers in FULL (use_pcs) space
  centers_use <- pts_use[sample.int(n, k), , drop = FALSE]
  cl_old <- integer(n)

  draw <- function(point_cols, main_txt) {
    s3d <- scatterplot3d::scatterplot3d(
      x = pts_plot[,1], y = pts_plot[,2], z = pts_plot[,3],
      color = point_cols, pch = pch, cex.symbols = cex,
      xlab = paste0("PC", plot_pcs[1]),
      ylab = paste0("PC", plot_pcs[2]),
      zlab = paste0("PC", plot_pcs[3]),
      main = main_txt,
      angle = theta
    )

    # centers need to be shown in plot space too:
    # take their coordinates from centers_use but only those PCs that are plotted.
    # If plot_pcs is not a subset of use_pcs, we must map them via the original pcx columns.
    centers_plot <- matrix(NA_real_, nrow = k, ncol = 3)
    for (j in 1:k) {
      # center in full PC coordinates (all columns) isn't directly stored,
      # so reconstruct by averaging in original pcx columns for assigned members later.
      # During init, centers_use lives only in use_pcs; so we can still plot by
      # taking those dimensions if plot_pcs are included; otherwise plot after first assignment.
      centers_plot[j,] <- NA_real_
    }

    invisible(s3d)
  }

  # initial view: points only + seed markers if possible
  s3d <- scatterplot3d::scatterplot3d(
    x = pts_plot[,1], y = pts_plot[,2], z = pts_plot[,3],
    color = "grey70", pch = pch_vec, cex.symbols = cex,
    xlab = paste0("PC", plot_pcs[1]),
    ylab = paste0("PC", plot_pcs[2]),
    zlab = paste0("PC", plot_pcs[3]),
    main = "init seeds (press Enter)",
    angle = theta
  )

  # plot initial centers only if plot_pcs are included in use_pcs
  if (all(plot_pcs %in% use_pcs)) {
    # find which columns inside pts_use correspond to the plotted PCs
    idx_in_use <- match(plot_pcs, use_pcs)
    centers_plot <- centers_use[, idx_in_use, drop = FALSE]
    cc <- s3d$xyz.convert(centers_plot[,1], centers_plot[,2], centers_plot[,3])
    points(cc$x, cc$y, pch = center_pch, cex = center_cex, lwd = center_lwd, col = cols)
  }

  if (pause) readline("Enter to continue...")

  for (t in 1:iters) {
    # assign in FULL (use_pcs) space
    d2 <- sapply(1:k, function(j) {
      rowSums((pts_use - matrix(centers_use[j,], nrow = n, ncol = ncol(pts_use), byrow = TRUE))^2)
    })
    cl <- max.col(-d2)

    # update centers in FULL (use_pcs) space (with empty-cluster reseed)
    for (j in 1:k) {
      idx <- which(cl == j)
      if (length(idx) > 0) centers_use[j,] <- colMeans(pts_use[idx, , drop = FALSE])
      else centers_use[j,] <- pts_use[sample.int(n, 1), ]
    }

    # redraw in PLOT (3D) space
    s3d <- scatterplot3d::scatterplot3d(
      x = pts_plot[,1], y = pts_plot[,2], z = pts_plot[,3],
      color = cols[cl], pch = pch_vec, cex.symbols = cex,
      xlab = paste0("PC", plot_pcs[1]),
      ylab = paste0("PC", plot_pcs[2]),
      zlab = paste0("PC", plot_pcs[3]),
      main = paste("iter", t, "(distance uses PCs:", length(use_pcs), ")"),
      angle = theta
    )

    # project centers to plot space when possible
    if (all(plot_pcs %in% use_pcs)) {
      idx_in_use <- match(plot_pcs, use_pcs)
      centers_plot <- centers_use[, idx_in_use, drop = FALSE]
    } else {
      # If plotted PCs aren’t included in use_pcs, best fallback is to compute centers
      # in full original PC space from current assignments:
      centers_plot <- t(vapply(1:k, function(j) {
        idx <- which(cl == j)
        if (length(idx) > 0) colMeans(pcx[idx, plot_pcs, drop = FALSE]) else pcx[sample.int(n,1), plot_pcs]
      }, numeric(3)))
    }

    cc <- s3d$xyz.convert(centers_plot[,1], centers_plot[,2], centers_plot[,3])
    points(cc$x, cc$y, pch = center_pch, cex = center_cex, lwd = center_lwd, col = cols)

    if (pause) readline("Enter to continue...")

    if (identical(cl, cl_old)) break
    cl_old <- cl
  }

  invisible(list(cluster = cl_old, centers_use = centers_use,
                 use_pcs = use_pcs, plot_pcs = plot_pcs))
}
```

What should k be? We know there are 4 population groups here. How does the algorithm do at calling them?

```{webr}
        axes = FALSE, main = main)
  axis(1, at = seq(0,1,length.out=ncol(m)), labels = colnames(m))
  axis(2, at = seq(0,1,length.out=nrow(m)), labels = rev(rownames(m)))
  # overlay counts
  for (i in 1:nrow(m)) for (j in 1:ncol(m)) {
    # map (i,j) to image coordinates: j along x, reversed i along y
    x <- (j-1)/(ncol(m)-1)
    y <- (nrow(m)-i)/(nrow(m)-1)
```

```{webr}
# ---- helper: majority vote with tie-break by nearest-neighbor (distance) ----
knn_vote <- function(labels, dists) {
  tab <- sort(table(labels), decreasing = TRUE)
  top_n <- as.integer(tab[1])
  tied <- names(tab)[tab == top_n]
  if (length(tied) == 1) return(tied)
  # tie-break: among tied classes, pick the class of the single closest neighbor
  labels[which.min(dists)]
}
```
